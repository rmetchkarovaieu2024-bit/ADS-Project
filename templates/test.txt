    <script> # javasctipt code
        document.getElementById('save-btn').addEventListener('click', () => {
          const data = {};
            document.querySelectorAll('.subject-item').forEach((el, idx) => {
            const subj = el.querySelector('.subject-name').textContent;
            const sel  = el.querySelector('select');
            data[subj] = parseInt(sel.value, 10);
          });

          fetch('/save_weights', {
            method: 'POST',
            headers: {'Content-Type': 'application/json'},
            body: JSON.stringify(data)
          })
          .then(r => r.json())
          .then(res => alert(res.message || res.error))
          .catch(_ => alert('Save failed'));
        });
    </script>
